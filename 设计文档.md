一、项目概述
1.1 项目目标
开发一款基于 FSRS（Free Spaced Repetition Scheduler）算法的闪卡记忆应用，支持 iOS 和 Android 双端，提供美观流畅的 UI 交互体验，并具备多设备同步、云端备份、AI 智能生成卡片、开放 API 等能力。

- 主体：瀑布流/列表 展示牌组卡片，每个卡片显示名称、进度环、今日待复习数

复习页（核心交互）

- 中央：卡片区域，支持点击翻转(3D 翻转动画)
- 底部：四个评分按钮 (Again / Hard / Good / Easy)
- 评分后：卡片向上滑出 + 下一张从底部滑入
- 完成时：Lottie 庆祝动画 + 统计摘要
统计页
- 热力图日历：展示每日复习量 (类 GitHub 贡献图)
- 折线图：记忆保持率趋势

七、FSRS 算法详细实现 (可以采用开源实现方式)
7.1 FSRS-5 算法概述
FSRS（Free Spaced Repetition Scheduler）由 open-spaced-repetition 团队开发，基于 DSR（Difficulty-Stability-Retrievability）三变量记忆模型。相比 Anki 默认的 SM-2 算法，FSRS 具有更高的记忆预测精度。
7.2 核心公式
7.2.1 可提取性（Retrievability）
衡量某张卡片在给定时刻被正确回忆的概率：
R(t, S) = (1 + FACTOR * t / S) ^ (-DECAY)

其中：
  DECAY  = -0.5
  FACTOR = 19^(1/DECAY) - 1 ≈ 0.9
  t      = 距上次复习的天数 (elapsed_days)
  S      = 稳定性 (stability)
当 R = 0.9 时，t = S，即稳定性的含义是"记忆保持率降到 90% 所需的天数"。
7.2.2 初始难度（Initial Difficulty）
新卡片首次评分后的初始难度：
D0(G) = w4 - e^(w5 * (G - 1)) + 1

其中：
  G     = 评分 (1=Again, 2=Hard, 3=Good, 4=Easy)
  w4,w5 = 可训练权重
7.2.3 初始稳定性（Initial Stability）
新卡片首次评分后的初始稳定性：
S0(G) = w[G-1]

即直接取 w0, w1, w2, w3 作为四种评分对应的初始稳定性。
7.2.4 难度更新
D'(D, G) = w7 * D0(3) + (1 - w7) * (D - w6 * (G - 3))

其中 D0(3) 是 "Good" 评分的初始难度，作为均值回归锚点。
D 被 clamp 到 [1, 10] 范围内。
7.2.5 稳定性更新（成功回忆 / Rating ≥ 2）
S'_recall(D, S, R, G) = S * (e^(w8) * (11 - D) * S^(-w9) * (e^(w10 * (1 - R)) - 1) * hard_penalty * easy_bonus + 1)

其中：
  hard_penalty = w15  (当 G = 2 时)，否则 = 1
  easy_bonus   = w16  (当 G = 4 时)，否则 = 1
7.2.6 稳定性更新（遗忘 / Rating = 1）
S'_forget(D, S, R) = w11 * D^(-w12) * ((S + 1)^w13 - 1) * e^(w14 * (1 - R))

import 'dart:math';

/// FSRS-5 算法引擎
class FSRSEngine {
  final List<double> w;  // 19 个权重参数
  static const double DECAY = -0.5;
  static final double FACTOR = pow(19, 1 / DECAY) - 1;  // ≈ 0.9

  FSRSEngine({List<double>? weights})
      : w = weights ?? defaultWeights;

  /// 计算可提取性 R
  double retrievability(double elapsedDays, double stability) {
    if (stability <= 0) return 0;
    return pow(1 + FACTOR * elapsedDays / stability, DECAY).toDouble();
  }

  /// 新卡初始稳定性
  double initStability(int rating) {
    return max(w[rating - 1], 0.1);
  }

  /// 新卡初始难度
  double initDifficulty(int rating) {
    double d = w[4] - exp(w[5] * (rating - 1)) + 1;
    return d.clamp(1.0, 10.0);
  }

  /// 更新难度
  double nextDifficulty(double d, int rating) {
    double d0 = initDifficulty(3);  // Good 作为均值回归锚点
    double newD = w[7] * d0 + (1 - w[7]) * (d - w[6] * (rating - 3));
    return newD.clamp(1.0, 10.0);
  }

  /// 成功回忆时更新稳定性
  double nextStabilityOnRecall(double d, double s, double r, int rating) {
    double hardPenalty = (rating == 2) ? w[15] : 1.0;
    double easyBonus = (rating == 4) ? w[16] : 1.0;
    double newS = s * (exp(w[8]) * (11 - d) * pow(s, -w[9]) *
        (exp(w[10] * (1 - r)) - 1) * hardPenalty * easyBonus + 1);
    return max(newS, 0.1);
  }

  /// 遗忘时更新稳定性 (rating = 1)
  double nextStabilityOnForget(double d, double s, double r) {
    double newS = w[11] * pow(d, -w[12]) * (pow(s + 1, w[13]) - 1) *
        exp(w[14] * (1 - r));
    return max(min(newS, s), 0.1);  // 遗忘后稳定性不应超过之前
  }

  /// 计算下次复习间隔（天数）
  double nextInterval(double stability, {double desiredRetention = 0.9}) {
    double interval = (stability / FACTOR) *
        (pow(desiredRetention, 1 / DECAY) - 1);
    return max(interval, 1.0);
  }

  /// 核心方法：处理一次复习，返回更新后的状态
  FSRSState review(FSRSState state, int rating, DateTime now) {
    double elapsedDays = state.lastReviewAt != null
        ? now.difference(state.lastReviewAt!).inMinutes / 1440.0
        : 0;

    double newD, newS;
    int newState;

    if (state.state == 0) {
      // === 新卡首次复习 ===
      newD = initDifficulty(rating);
      newS = initStability(rating);
      newState = (rating == 1) ? 1 : (rating == 4) ? 2 : 1;
    } else if (state.state == 1 || state.state == 3) {
      // === Learning / Relearning ===
      newD = nextDifficulty(state.difficulty, rating);
      if (rating == 1) {
        newS = state.stability;  // Again 不改变稳定性
        newState = state.state;   // 停留在当前状态
      } else if (rating == 4 || (rating == 3 && state.state == 3)) {
        // Easy 或 Relearning 的 Good → 进入 Review
        double r = retrievability(elapsedDays, state.stability);
        newS = nextStabilityOnRecall(state.difficulty, state.stability, r, rating);
        newState = 2;
      } else {
        newS = state.stability;
        newState = state.state;
      }
    } else {
      // === Review 状态 ===
      double r = retrievability(elapsedDays, state.stability);
      newD = nextDifficulty(state.difficulty, rating);
      if (rating == 1) {
        newS = nextStabilityOnForget(state.difficulty, state.stability, r);
        newState = 3;  // 进入 Relearning
      } else {
        newS = nextStabilityOnRecall(state.difficulty, state.stability, r, rating);
        newState = 2;
      }
    }

    // 计算下次复习时间
    DateTime dueDate;
    if (newState == 1 || newState == 3) {
      // Learning/Relearning: 短期调度
      dueDate = _shortTermSchedule(newState, rating, now);
    } else {
      // Review: 长期调度
      double interval = nextInterval(newS);
      dueDate = now.add(Duration(minutes: (interval * 1440).round()));
    }

    return FSRSState(
      state: newState,
      difficulty: newD,
      stability: newS,
      retrievability: retrievability(0, newS),
      dueDate: dueDate,
      lastReviewAt: now,
      reps: state.reps + 1,
      lapses: state.lapses + (rating == 1 ? 1 : 0),
      elapsedDays: elapsedDays,
      scheduledDays: dueDate.difference(now).inMinutes / 1440.0,
    );
  }

  DateTime _shortTermSchedule(int state, int rating, DateTime now) {
    Map<int, Duration> learningIntervals = {
      1: Duration(minutes: 1),   // Again
      2: Duration(minutes: 5),   // Hard
      3: Duration(minutes: 10),  // Good
    };
    Map<int, Duration> relearningIntervals = {
      1: Duration(minutes: 10),  // Again
      2: Duration(minutes: 10),  // Hard
    };
    Duration interval;
    if (state == 1) {
      interval = learningIntervals[rating] ?? Duration(minutes: 10);
    } else {
      interval = relearningIntervals[rating] ?? Duration(minutes: 10);
    }
    return now.add(interval);
  }
}

/// FSRS 状态数据类
class FSRSState {
  final int state;            // 0:New, 1:Learning, 2:Review, 3:Relearning
  final double difficulty;
  final double stability;
  final double retrievability;
  final DateTime dueDate;
  final DateTime? lastReviewAt;
  final int reps;
  final int lapses;
  final double elapsedDays;
  final double scheduledDays;

  const FSRSState({
    this.state = 0,
    this.difficulty = 0,
    this.stability = 0,
    this.retrievability = 0,
    required this.dueDate,
    this.lastReviewAt,
    this.reps = 0,
    this.lapses = 0,
    this.elapsedDays = 0,
    this.scheduledDays = 0,
  });
}

9.3 LLM Prompt 设计
你是一个专业的教育内容设计师，擅长将知识提炼为高效的闪卡（Flashcard）。

## 任务
根据用户提供的内容，生成用于间隔重复学习的闪卡。

## 要求
1. 每张卡片包含 front（问题）和 back（答案）
2. 问题应考察核心概念、关键定义、重要关系，避免过于琐碎的细节
3. 答案简洁精确，控制在 1-3 句话以内
4. 确保问题有唯一确定的答案，避免主观性问题
5. 为每张卡片添加 1-3 个相关 tags
6. 难度级别：{difficulty_hint}
7. 生成语言：{language}
8. 最多生成 {card_count} 张卡片

## 输出格式
严格输出 JSON 数组，不要包含任何其他内容：
[
  {
    "front": "问题内容",
    "back": "答案内容",
    "tags": ["标签1", "标签2"]
  }
]

## 用户提供的内容
{content}
9.4 后端处理流程
func (s *AIService) GenerateCards(ctx context.Context, req AIGenerateRequest) ([]CardDTO, error) {
    // 1. 预处理：提取文本
    var text string
    switch req.InputType {
    case "text":
        text = req.Content
    case "pdf":
        text = s.pdfExtractor.Extract(req.Content) // 从上传的 PDF 提取文本
    case "url":
        text = s.webScraper.ExtractArticle(req.Content) // 从 URL 提取正文
    }

    // 2. 文本分段（超过 LLM 上下文窗口时）
    chunks := s.chunker.Split(text, maxChunkSize)

    // 3. 逐段调用 LLM 生成卡片
    var allCards []CardDTO
    for _, chunk := range chunks {
        prompt := buildPrompt(chunk, req)
        response, err := s.llmClient.Chat(ctx, prompt)
        if err != nil {
            return nil, err
        }
        cards := parseCardsFromJSON(response)
        allCards = append(allCards, cards...)
    }

    // 4. 后处理：去重 + 质量过滤
    allCards = s.deduplicator.Deduplicate(allCards)
    allCards = s.qualityFilter.Filter(allCards)

    // 5. 截取到请求数量
    if len(allCards) > req.CardCount {
        allCards = allCards[:req.CardCount]
    }

    return allCards, nil
}

client.ai.confirm(deck_id=deck.id, cards=ai_cards)
